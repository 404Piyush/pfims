pythondef get_investment_recommendation(answers):
    # STEP 1: Check blockers first
    if answers['high_interest_debt']:
        return {
            'status': 'blocked',
            'message': 'Pay off high-interest debt first',
            'recommendation': 'debt_payoff_plan'
        }
    
    # STEP 2: Calculate risk score
    score = (
        answers['risk_tolerance'] * 0.35 +
        answers['duration'] * 0.30 +
        answers['savings_capacity'] * 0.20 +
        answers['financial_goal'] * 0.15
    )
    
    # STEP 3: Determine base profile
    if score <= 2.0:
        profile = "Conservative"
        allocation = {
            'large_cap': 40,
            'mid_cap': 10,
            'small_cap': 0,
            'bonds': 50
        }
    elif score <= 3.0:
        profile = "Moderate"
        allocation = {
            'large_cap': 40,
            'mid_cap': 25,
            'small_cap': 10,
            'bonds': 25
        }
    else:
        profile = "Aggressive"
        allocation = {
            'large_cap': 35,
            'mid_cap': 30,
            'small_cap': 25,
            'bonds': 10
        }
    
    # STEP 4: Apply age adjustments
    if answers['age'] > 50:
        allocation['small_cap'] = min(allocation['small_cap'], 15)
        allocation['bonds'] = max(allocation['bonds'], 30)
        # Redistribute the difference
        diff = 30 - allocation['bonds']
        if diff > 0:
            allocation['large_cap'] -= diff
            allocation['bonds'] = 30
    
    if answers['age'] < 25:
        total_equity = allocation['large_cap'] + allocation['mid_cap'] + allocation['small_cap']
        if total_equity < 70:
            # Increase equity to at least 70%
            increase = 70 - total_equity
            allocation['bonds'] -= increase
            allocation['large_cap'] += increase
    
    # STEP 5: Emergency fund override
    if not answers['has_emergency_fund']:
        allocation['liquid_fund'] = 20
        # Reduce others proportionally
        for key in ['large_cap', 'mid_cap', 'small_cap', 'bonds']:
            allocation[key] = int(allocation[key] * 0.8)
    
    # STEP 6: Return recommendation
    return {
        'status': 'success',
        'profile': profile,
